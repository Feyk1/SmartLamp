# Smart Lamp Demo — firmware/test/unit
#
# Firmware unit tests. Each test is an executable registered in CTest.

add_executable(test_flash test_flash.c)

target_link_libraries(test_flash PRIVATE flash_hal unity)
target_compile_features(test_flash PRIVATE c_std_99)

add_test(NAME test_flash COMMAND test_flash)
set_tests_properties(test_flash PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")

# WS transport: EINTR regression caused by the POSIX port tick signal.
add_executable(test_net_transport
    test_net_transport.c
    "${CMAKE_SOURCE_DIR}/firmware/src/comms/net_transport_posix.c")

target_compile_features(test_net_transport PRIVATE c_std_99)
target_compile_definitions(test_net_transport PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(test_net_transport PRIVATE
    "${CMAKE_SOURCE_DIR}/firmware/include")
target_link_libraries(test_net_transport PRIVATE unity Threads::Threads)

add_test(NAME test_net_transport COMMAND test_net_transport)
set_tests_properties(test_net_transport PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
